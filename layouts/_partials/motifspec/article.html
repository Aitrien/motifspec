{{- /* The cx-article layout: TOC before the main column in the DOM, then prose ending in the flower end mark. Context: dict "page" "toc" (always | auto | never). auto shows the TOC when the content has three or more h2s (table-of-contents.md); a case study passes always. A documentation site (params.motifspec.layout "documentation") has no end mark: its only ornament is the footer band (docs/layouts/documentation.md). */ -}} {{- $p := .page -}} {{- $where := "" -}} {{- with $p.File }}{{ $where = printf "content/%s" .Path }}{{ else }}{{ $where = $p.Path }}{{ end -}} {{- $docs := eq site.Params.motifspec.layout "documentation" -}} {{- $h2s := partial "motifspec/h2s.html" $p -}} {{- $toc := and (ne .toc "never") (or (eq .toc "always") (ge (len $h2s) 3)) -}} {{- if and (eq .toc "always") (lt (len $h2s) 1) }}{{ errorf "%s: toc \"always\" needs h2 sections to list (table-of-contents.md)." $where }}{{ end -}} {{- /* ornament.md: the flower ("g") goes after the last sentence, so the content must end with a paragraph. */ -}} {{- $content := $p.Content | strings.TrimSpace -}} {{- if gt (len (findRE `class="cx-callout[ "]` $content)) 2 }}{{ errorf "%s: more than two callouts in one article; they lose force when repeated (callout.md)." $where }}{{ end -}} {{- $endMark := `g

` -}} {{- if $docs -}} {{- else if strings.HasSuffix $content "

" -}} {{- $content = printf "%s%s" (strings.TrimSuffix "

" $content) $endMark -}} {{- else -}} {{- warnf "%s: the article should end with a paragraph so the end mark follows the last sentence." $where -}} {{- end }}
{{- if $toc }} {{ partial "motifspec/table-of-contents.html" $p }} {{- end }}
