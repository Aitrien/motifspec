{{- /* pagination.md between entries of one collection, oldest = 1. Previous is the older entry, Next the newer. A missing direction stays as a disabled cell. Context: dict "page" "label" ("Post navigation") "noun" ("Post"). */ -}} {{- $p := .page -}} {{- $all := (where site.RegularPages "Section" $p.Section).ByDate -}} {{- $at := 0 }}{{ range $i, $e := $all }}{{ if eq $e.RelPermalink $p.RelPermalink }}{{ $at = $i }}{{ end }}{{ end -}} {{- $prev := cond (gt $at 0) (index $all (sub $at 1)) nil -}} {{- $next := cond (lt (add $at 1) (len $all)) (index $all (add $at 1)) nil -}} {{- if gt (len $all) 1 }}
{{- with $prev }} Previous {{ .Title }} {{- else }} Previous First entry {{- end }}

{{ .noun }}{{ add $at 1 }} / {{ len $all }}

{{- with $next }} Next {{ .Title }} {{- else }} Next Latest entry {{- end }}
{{- end -}}
